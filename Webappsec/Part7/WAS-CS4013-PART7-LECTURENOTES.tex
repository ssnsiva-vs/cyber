\section{Approach to Web application security}

\begin{frame}
\frametitle{Introduction}
Broad areas of Web application security 
\begin{itemize}
	\item Minimum security features
	\item Software Development Lifecycle
	\item Secure coding practices
	\item Testing
\end{itemize}


\end{frame}

\begin{frame}
	\frametitle{Minimum Security features}
	
	\begin{itemize}
		\item User management (Authentication, Authorization, Access Control)
		\item User Input management
		\item Data Security
		\item Attacks and Errors prevention
		\item Application infrastructure security
	\end{itemize}
	
	
\end{frame}


\begin{frame}
	\frametitle{Software Development Lifecycle}
SDLC Perspective
	\begin{itemize}
		\item Design, Coding, Testing and release stages
		\item The stages are sequential
		\item Every coding project contains bugs, a percentage of which has the potential to get converted into a vulnerability
		\item Bugs when caught early in the development stage is less painful to fix, compared with catching them during testing or in the release phase.
	\end{itemize}
	
	
\end{frame}

\begin{frame}
	\frametitle{Secure coding practices}
\url{https://cheatsheetseries.owasp.org/index.html}
	\begin{itemize}
		\item Authentication, Authorization, Session management
		\item JWT, OAUTH2 and SAML
		\item IDE plugins
		\item Zero trust
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Testing for errors and vulnerabilities}

		\item SAST and DAST
		\item Vulnerability penetration testing
		\item Prioritization and fixes
		\item Regression
	\end{itemize}
\end{frame}
